# ============================================================
# VOICE ANALYSIS FUNCTION
# ============================================================

def analyze_voice(
    start_time,
    end_time
):

    """
    Analyze the voice in one Whisper segment.

    Returns:

        gender
        median_pitch
        confidence
    """

    # --------------------------------------------------------
    # Convert time to samples
    # --------------------------------------------------------

    start_sample = int(
        max(
            0,
            start_time
        ) * sample_rate
    )

    end_sample = int(
        min(
            len(audio),
            end_time * sample_rate
        )
    )


    segment_audio = audio[
        start_sample:end_sample
    ]


    # --------------------------------------------------------
    # Very short segment
    # --------------------------------------------------------

    if len(segment_audio) < (
        sample_rate * 0.15
    ):

        return (
            "unknown",
            0.0,
            0.0
        )


    # --------------------------------------------------------
    # Normalize
    # --------------------------------------------------------

    try:

        segment_audio = librosa.util.normalize(
            segment_audio
        )

    except Exception:

        return (
            "unknown",
            0.0,
            0.0
        )


    # --------------------------------------------------------
    # Pitch detection
    # --------------------------------------------------------

    try:

        f0, voiced_flag, voiced_probs = (
            librosa.pyin(
                segment_audio,

                fmin=65,

                fmax=350,

                sr=sample_rate,

                frame_length=2048,

                hop_length=256
            )
        )

    except Exception:

        return (
            "unknown",
            0.0,
            0.0
        )


    # --------------------------------------------------------
    # Remove NaN values
    # --------------------------------------------------------

    valid_pitch = f0[
        ~np.isnan(f0)
    ]


    if len(valid_pitch) == 0:

        return (
            "unknown",
            0.0,
            0.0
        )


    # --------------------------------------------------------
    # Remove extreme pitch values
    # --------------------------------------------------------

    valid_pitch = valid_pitch[
        (valid_pitch >= 65)
        &
        (valid_pitch <= 350)
    ]


    if len(valid_pitch) == 0:

        return (
            "unknown",
            0.0,
            0.0
        )


    # --------------------------------------------------------
    # Pitch statistics
    # --------------------------------------------------------

    median_pitch = float(
        np.median(
            valid_pitch
        )
    )

    average_pitch = float(
        np.mean(
            valid_pitch
        )
    )


    # --------------------------------------------------------
    # Voiced ratio
    # --------------------------------------------------------

    voiced_ratio = float(
        np.mean(
            voiced_flag
        )
    )


    # ========================================================
    # GENDER ESTIMATION
    # ========================================================

    if median_pitch <= MALE_PITCH_MAX:

        gender = "male"

    elif median_pitch >= FEMALE_PITCH_MIN:

        gender = "female"

    else:

        # ----------------------------------------------------
        # Borderline pitch
        # ----------------------------------------------------

        if average_pitch < 165:

            gender = "male"

        elif average_pitch > 170:

            gender = "female"

        else:

            gender = "unknown"


    # ========================================================
    # CONFIDENCE
    # ========================================================

    distance_from_boundary = min(
        abs(
            median_pitch
            -
            MALE_PITCH_MAX
        ),

        abs(
            median_pitch
            -
            FEMALE_PITCH_MIN
        )
    )


    pitch_confidence = min(
        distance_from_boundary / 35.0,
        1.0
    )


    confidence = round(
        (
            pitch_confidence * 0.7
        )
        +
        (
            voiced_ratio * 0.3
        ),
        3
    )


    return (
        gender,
        round(
            median_pitch,
            2
        ),
        confidence
    )

speech_segments = []


whisper_segments = result.get(
    "segments",
    []
)


if not whisper_segments:

    print()
    print(
        "❌ ERROR: Whisper returned no segments."
    )

    raise SystemExit(1)


# ============================================================
# LOOP THROUGH SENTENCES
# ============================================================

for number, segment in enumerate(
    whisper_segments,
    start=1
):

    text = segment.get(
        "text",
        ""
    ).strip()


    if not text:

        continue


    # --------------------------------------------------------
    # Timing
    # --------------------------------------------------------

    start = float(
        segment.get(
            "start",
            0
        )
    )


    end = float(
        segment.get(
            "end",
            0
        )
    )


    duration = end - start


    print()
    print("-" * 70)

    print(
        f"Segment {number}"
    )

    print(
        f"Time: {start:.2f} → {end:.2f}"
    )

    print(
        "Text:",
        text
    )


    # --------------------------------------------------------
    # Voice analysis
    # --------------------------------------------------------

    (
        gender,
        pitch,
        confidence
    ) = analyze_voice(
        start,
        end
    )


    print(
        "Gender:",
        gender
    )

    print(
        "Pitch:",
        pitch,
        "Hz"
    )

    print(
        "Confidence:",
        confidence
    )


    # ========================================================
    # CREATE SEGMENT OBJECT
    # ========================================================

    speech_segments.append(
        {

            "id":
                len(
                    speech_segments
                ) + 1,

            "start":
                round(
                    start,
                    3
                ),

            "end":
                round(
                    end,
                    3
                ),

            "duration":
                round(
                    duration,
                    3
                ),

            "text":
                text,

            "gender":
                gender,

            "pitch":
                pitch,

            "confidence":
                confidence

        }
    )
